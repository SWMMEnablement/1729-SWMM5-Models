@echo off
setlocal enabledelayedexpansion

:: Set the path to the SWMM5 executable (update this if it's not in your PATH or specify full path)
set SWMM_EXE=C:\Program Files (x86)\EPA SWMM 5.2.4\runswmm.exe

:: Set the input file name (assuming the same name in each subdirectory, e.g., model.inp)
set INPUT_FILE=model.inp

:: Set the report and output file suffixes (will be generated in each subdirectory)
set REPORT_SUFFIX=report.rpt
set OUTPUT_SUFFIX=out.out

:: Loop through all subdirectories in the current directory
for /d %%D in (*) do (
    if exist "%%D\%INPUT_FILE%" (
        echo Running SWMM in subdirectory: %%D
        pushd "%%D"
        %SWMM_EXE% "%INPUT_FILE%" "%REPORT_SUFFIX%" "%OUTPUT_SUFFIX%"
        if %ERRORLEVEL% neq 0 (
            echo Error running SWMM in %%D
        ) else (
            echo Successfully ran SWMM in %%D
        )
        popd
    ) else (
        echo No input file found in %%D
    )
)

echo All subdirectories processed.
pause
